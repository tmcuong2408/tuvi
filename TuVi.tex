\begin{abstract}
	Từ ngàn xưa, vận số được xem là một môn khoa học dự trắc cổ phương Đông, ứng dụng các quy luật thiên văn và lịch pháp để giải mã quỹ đạo đời người qua hệ thống lá số. Tuy nhiên, nếu gác lại lớp vỏ bọc huyền học và nhìn nhận dưới lăng khúc toán học hiện đại, hệ thống này thực chất là một mô hình mã hóa thông tin vô cùng chặt chẽ.
	
	Tài liệu này xây dựng và hệ thống hóa lại bộ môn vận số dưới góc độ toán học, từ đó định danh nó dưới một tên gọi mới: {Vận số học}.
	
	Trong không gian của Vận số học, mỗi lá số không phải là một định mệnh thần bí, mà là một tập hợp vũ trụ được cấu thành từ các tập con phân hoạch (hệ thống 12 cung địa bàn) và các phần tử tương tác (tập hợp các vì sao). Bằng cách sử dụng ngôn ngữ của Lý thuyết tập hợp—như phép ánh xạ từ thời điểm sinh vào không gian lá số, phép giao tạo nên các cách cục, phép hợp giữa các cung liên kết, hay hàm số mô tả sự dịch chuyển của vận hạn theo thời gian—Vận số học bóc tách toàn bộ lớp sương mù tín ngưỡng để trả về bản chất cốt lõi: một hệ thống thuật toán phân tích dữ liệu không gian - thời gian của cổ nhân, được số hóa và cấu trúc hóa dưới tư duy logic thuần túy.
\end{abstract}

\chapter{Nguyên lý âm dương}
\section{Tiên đề vũ trụ}

Gọi $\Omega$ là tập tất cả các vật trong vũ trụ. Ta đưa ra tiên đề hệ tiên đề :
\begin{enumerate}
	\item Tiên đề phân loại.Mọi vật $\omega \in \Omega$ thì $\omega \in \Omega_+ $ hoặc $ \omega \in \Omega_-$ và tồn tại một ánh xạ  $\Phi  : \Omega \to \{-1,1 \}$ thỏa mãn: $$\Phi(\omega) := \begin{cases}
		1 \text{ nếu } \omega \in \Omega_+ \\ 
		-1 \text{ nếu } \omega \in \Omega_- \\	
	\end{cases}
	$$
	\item  Tiên đề cân bằng: \[
	\sum_{\omega \in \Omega}^{}{\Phi(\omega)} = 0
	\]
	\item Tiên đề đầy đủ: \[\Omega = \Omega_- \cup \Omega_+\]
	\item Tiên đề tách: \[\Omega_- \cap \Omega_+	= \emptyset\]
	
\end{enumerate}
Trong đó 
\begin{align*}
	&\Omega_+:= \{\omega \in \Omega :  \omega \text{ mang tính dương}\} \\
	&\Omega_-:= \{\omega \in \Omega :  \omega \text{ mang tính âm}\} \\
\end{align*}
\begin{definition}[Độ phân giải vận số]
	Từ $n$ nguyên dương cho trước thì $\Omega$ có thể được phân hoạch thành $2^n$ phần rời nhau với $ n \in \mathbb{N^*}$. Ta gọi $n$ là độ phân giải vận số.
\end{definition}
\begin{remark}
	Xuyên suốt tài liệu này ta giả định độ phân giải vận số là $n$.
\end{remark}
\begin{definition}[Sao vận số]
	Mỗi sao được đánh số thứ tự bằng hệ nhị phân ký hiệu $V_\alpha$ với $\alpha \in \set{0,1}^n$. Cụ thể, chuỗi nhị phân $\alpha = (a_1, a_2, \dots, a_n) \in \{0, 1\}^n$ xác định một tập hợp phần tử trong không gian tích Descartes:$$V_\alpha := \Omega_{a_1} \times \Omega_{a_2} \times \dots \times \Omega_{a_n} = \prod_{i=1}^{n} \Omega_{a_i}$$thỏa mãn các điều kiện:
	\begin{enumerate}
		\item $\Omega_0 = \Omega_-, \Omega_1 = \Omega_+$.
		\item Tính rời nhau: $V_\alpha \cap V_\beta = \emptyset$ với mọi $\alpha \neq \beta$.
		\item Tính trọn vẹn: $\bigcup_{\alpha \in \{0, 1\}^n} V_\alpha = \Omega$.
	\end{enumerate}
\end{definition}
\begin{definition}[Tượng]
	Xét $\alpha \in \set{0,1}^n$, ta gọi mỗi phần tử $v_\alpha \in V_\alpha \in V$ là một tượng.
\end{definition}
\begin{definition}[Không gian sao]
	Tập tất cả các sao vận số ứng với độ phân giải $n$ được gọi là không gian sao, được ký hiệu là $V$. Dễ thấy $\card V = 2^n$.
\end{definition}
\begin{definition}[Hàm ngữ nghĩa]
	Xét $\alpha \in \set{0,1}^n$, ứng với từng $V_\alpha \in V$, ta định nghĩa $m : V_\alpha \to S$ và gọi $m$ là hàm ngữ nghĩa của $V_\alpha$, với $s \in S$ là một câu diễn dịch ứng với $v_\alpha \in V_\alpha$.
\end{definition}
